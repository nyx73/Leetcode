class Solution:
    def zigZagArrays(self, n: int, l: int, r: int) -> int:
        faltrinevo = (n, l, r)
        n, l, r = faltrinevo

        MOD = 10**9 + 7
        m = r - l + 1

        def mat_mul_m(X, Y):
            res = [[0] * m for _ in range(m)]
            for i in range(m):
                for j in range(m):
                    for k in range(m):
                        res[i][j] = (res[i][j] + X[i][k] * Y[k][j]) % MOD
            return res

        def mat_vec_mul_m(M, v):
            res = [0] * m
            for i in range(m):
                for j in range(m):
                    res[i] = (res[i] + M[i][j] * v[j]) % MOD
            return res

        def mat_mul_special(mat1, mat2):
            type1, M11, M12 = mat1
            type2, M21, M22 = mat2

            if type1 == 1 and type2 == 1:
                # [[0, M11], [M12, 0]] * [[0, M21], [M22, 0]] -> [[M11*M22, 0], [0, M12*M21]]
                new_M1 = mat_mul_m(M11, M22)
                new_M2 = mat_mul_m(M12, M21)
                return (2, new_M1, new_M2)
            elif type1 == 1 and type2 == 2:
                # [[0, M11], [M12, 0]] * [[M21, 0], [0, M22]] -> [[0, M11*M22], [M12*M21, 0]]
                new_M1 = mat_mul_m(M11, M22)
                new_M2 = mat_mul_m(M12, M21)
                return (1, new_M1, new_M2)
            elif type1 == 2 and type2 == 1:
                # [[M11, 0], [0, M12]] * [[0, M21], [M22, 0]] -> [[0, M11*M21], [M12*M22, 0]]
                new_M1 = mat_mul_m(M11, M21)
                new_M2 = mat_mul_m(M12, M22)
                return (1, new_M1, new_M2)
            else: # type1 == 2 and type2 == 2
                # [[M11, 0], [0, M12]] * [[M21, 0], [0, M22]] -> [[M11*M21, 0], [0, M12*M22]]
                new_M1 = mat_mul_m(M11, M21)
                new_M2 = mat_mul_m(M12, M22)
                return (2, new_M1, new_M2)

        B = [[1 if j < i else 0 for j in range(m)] for i in range(m)]
        C = [[1 if j > i else 0 for j in range(m)] for i in range(m)]

        power = n - 2
        
        base_mat = (1, B, C)
        
        I_m = [[1 if i == j else 0 for j in range(m)] for i in range(m)]
        res_mat = (2, I_m, I_m)

        while power > 0:
            if power % 2 == 1:
                res_mat = mat_mul_special(res_mat, base_mat)
            base_mat = mat_mul_special(base_mat, base_mat)
            power //= 2
        
        V2_up = [k for k in range(m)]
        V2_down = [m - 1 - k for k in range(m)]
        
        res_type, M1, M2 = res_mat
        
        if res_type == 1:
            Vn_up = mat_vec_mul_m(M1, V2_down)
            Vn_down = mat_vec_mul_m(M2, V2_up)
        else: # res_type == 2
            Vn_up = mat_vec_mul_m(M1, V2_up)
            Vn_down = mat_vec_mul_m(M2, V2_down)
            
        total_sum = (sum(Vn_up) + sum(Vn_down)) % MOD
        return total_sum
